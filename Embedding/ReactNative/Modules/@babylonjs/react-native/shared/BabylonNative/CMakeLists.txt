set(BABYLON_NATIVE_BUILD_APPS OFF)
set(BABYLON_NATIVE_INSTALL OFF)
set(BABYLON_NATIVE_PLUGIN_TESTUTILS OFF)
set(BABYLON_NATIVE_USE_WINDOWS_APP_SDK OFF)
set(BABYLON_NATIVE_DISABLE_WEBMIN ON)
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/Repo/CMakeLists.txt")
    set(FETCHCONTENT_FULLY_DISCONNECTED ON)
    set(FETCHCONTENT_SOURCE_DIR_BX "${CMAKE_CURRENT_SOURCE_DIR}/deps/bx-src")
    set(FETCHCONTENT_SOURCE_DIR_BIMG "${CMAKE_CURRENT_SOURCE_DIR}/deps/bimg-src")
    set(FETCHCONTENT_SOURCE_DIR_BGFX "${CMAKE_CURRENT_SOURCE_DIR}/deps/bgfx-src")
    set(FETCHCONTENT_SOURCE_DIR_ARCANA.CPP "${CMAKE_CURRENT_SOURCE_DIR}/deps/arcana.cpp-src")
    set(FETCHCONTENT_SOURCE_DIR_SPIRV-CROSS "${CMAKE_CURRENT_SOURCE_DIR}/deps/spirv-cross-src")
    set(FETCHCONTENT_SOURCE_DIR_GLSLANG "${CMAKE_CURRENT_SOURCE_DIR}/deps/glslang-src")
    set(FETCHCONTENT_SOURCE_DIR_BASE-N "${CMAKE_CURRENT_SOURCE_DIR}/deps/base-n-src")
    set(FETCHCONTENT_SOURCE_DIR_IOS-CMAKE "${CMAKE_CURRENT_SOURCE_DIR}/deps/ios-cmake-src")
    set(FETCHCONTENT_SOURCE_DIR_ARCORE-ANDROID-SDK "${CMAKE_CURRENT_SOURCE_DIR}/deps/arcore-android-sdk-src")
    set(FETCHCONTENT_SOURCE_DIR_ASIO "${CMAKE_CURRENT_SOURCE_DIR}/deps/asio-src")
    set(FETCHCONTENT_SOURCE_DIR_LLHTTP "${CMAKE_CURRENT_SOURCE_DIR}/deps/llhttp-src")
    set(FETCHCONTENT_SOURCE_DIR_GSL "${CMAKE_CURRENT_SOURCE_DIR}/deps/gsl-src")
    set(FETCHCONTENT_SOURCE_DIR_METAL-CPP "${CMAKE_CURRENT_SOURCE_DIR}/deps/metal-cpp-src")

    set(BABYLON_NATIVE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/Repo")
else()
    set(BABYLON_NATIVE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}")
    while(NOT EXISTS "${BABYLON_NATIVE_SOURCE_DIR}/CMakeLists.txt"
        OR NOT EXISTS "${BABYLON_NATIVE_SOURCE_DIR}/Core/Graphics/CMakeLists.txt"
        OR NOT EXISTS "${BABYLON_NATIVE_SOURCE_DIR}/Dependencies/CMakeLists.txt")
        get_filename_component(_parent "${BABYLON_NATIVE_SOURCE_DIR}" DIRECTORY)
        if(_parent STREQUAL BABYLON_NATIVE_SOURCE_DIR)
            message(FATAL_ERROR "Cannot locate the Babylon Native checkout. Installed npm packages must include shared/BabylonNative/Repo.")
        endif()
        set(BABYLON_NATIVE_SOURCE_DIR "${_parent}")
    endwhile()
endif()

# Keep the binary layout identical for the podspec and MSBuild library paths.
add_subdirectory("${BABYLON_NATIVE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/Repo")
